/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：程序判断一个字符串中的圆括号是否平衡匹配。所谓的平衡匹配是指一个
      字符串中的左、右圆括号同时满足下列两个条件：
      1、整个字符串中左括号和右括号个数相等。
      2、每个右括号的左侧所有括号（连同该右括号），左括号的个数应
         大于或等于右括号的个数。

例如：字符串"(a)+(b)"中的圆括号是平衡匹配的。
      字符串"(a)+b)+(x+(y) "中的圆括号不平衡匹配(不满足第2个条件)
      函数check(char s[])的功能是检查字符串s中的圆括号是否平衡匹配，
      若平衡匹配则函数返回1，否则函数返回0。

程序运行结果如下：
        (((a+b)-c)*d)-e                   yes
        ((())())()                        yes
        ((()                              no
        )()(                              no
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <string.h>

#define  N 4

int check(char s[])
{
    char stack[100];
    int i=0,left;
    left=0;
/***********FOUND***********/
    while(stack[left]!='\0')
    {
        if(s[i]=='(')
            stack[left++]='(';
        if(s[i]==')')
/***********FOUND***********/
            if(stack[left])
                left--;
            else
                return 0;
        i++;
    }
/***********FOUND***********/
    return 1;
}
int main(void)
{
    char a[N][100]={"(((a+b)-c)*d)-e","((())())()","((()",")()("};
    int i,j;
    for(i=0;i<N;i++)
    {
        j=check(a[i]);
        if(j)
            printf("%-20s  %s\n",a[i],"yes");
        else
            printf("%-20s  %s\n",a[i],"no");
    }
}
