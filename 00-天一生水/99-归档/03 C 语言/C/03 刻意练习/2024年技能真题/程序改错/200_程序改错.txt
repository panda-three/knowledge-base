/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：设A-Z(a-z)的序号为1-26。函数void encode(char s[])的功能是：
      对字符串s中的字母加密处理，方法如下：
      1、将字符s[i]对应的字母的序号转换为5位二进制数x。
      2、取二进制数x的反序数得到y，并将y再转换成十进制数k。
      3、如果k小于等于26，则序号k对应的字母即为s[i]加密后的字母。
      4、如果k大于26，则取k%26的值为序号对应的字母即为s[i]加密后的字母。

例如：字符'a'的序号为1，二进制数为00001，反序数为10000，十进制数为16，
      则字母a加密成序号为16的字母p。
      字符'K'的序号是11，二进制数为01011，反序数为11010，十进制数为26，
      则字母K加密成序号为26的字母Z。
      字符'G'的序号是7，二进制数为00111，反序数是11100，十进制数为28，
      取28%26的值为2，字母G加密成序号为2的字母B。

程序运行结果如下：

加密前原字符串为：AKcDeFdG?
加密后的字符串为：PZxDtLdB?
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
-----------------------------------------------------------------------*/
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

void encode(char s[])
{
    int i,j,k,t[5],v,m;
    for(i=0;s[i]!='\0';i++)
    {
        if(isalpha(s[i]))
        {
            if(toupper(s[i])==s[i])
                k=s[i]-'A'+1;
            else
                k=s[i]-'a'+1;
            for(j=0;j<5;j++)
            {
                t[j]=k%2;
           /***********FOUND***********/
                k=k/10;
            }
            for(j=0;j<2;j++)
            {
          /***********FOUND***********/
                m=5-j;
                v=t[j];t[j]=t[m];t[m]=v;
            }
            for(k=0,j=4;j>=0;j--)
                k=k*2+t[j];
        /***********FOUND***********/
            k=k%26;
            if(toupper(s[i])==s[i])
                s[i]='A'+k-1;
            else
                s[i]='a'+k-1;
        }
    }
}
int main(void)
{
    char s[80];
    strcpy(s,"AKcDeFdG?");
    printf("加密前原字符串为：%s\n",s);
    encode(s);
    printf("加密后的字符串为：%s\n",s);
}
