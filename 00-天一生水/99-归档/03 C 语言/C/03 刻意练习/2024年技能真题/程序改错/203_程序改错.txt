/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：为了测试学生对英文单词的掌握程度，程序在给出的五个单词中随机
      抽取一个单词，用"*"替换该单词中的两个随机位置上的字母并显示替换后
      的单词，再将被测学生输入的单词与正确单词比较，显示对错信息。
      如果输入正确，显示“Right!”，如果输入错误，显示“Wrong!”，并给
      出正确答案。

测试数据与运行结果： 程序运行结果是随机的，即每次运行结果可能不同。
一次程序运行结果如下：
显示：The word is: fu*cti*n,  Please input correct word:
输入：function
显示：Right!
另一次程序运行结果如下：
显示：The word is: i*clud* ,  Please input correct word:
输入：inelude
显示：Wrong!  The word is : include

------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <time.h>

int funtest(char cp[][20],int i)
{
    int ix=0,iy=0,ik=0;
    char ca[20]={0},cb[20]={0};
    strcpy(ca,cp[i]);
    ik=strlen(ca);
    ix=rand()%ik;
    iy=rand()%ik;
/***********FOUND***********/
    if(ix!=iy)
        iy=rand()%ik;
    ca[ix]='*';
    ca[iy]='*';
    printf("The word is:%s,  ",ca);
    printf("Please input correct word:");
    gets(cb);
/***********FOUND***********/
    if(cb==cp[i])
        return 1;
    else
        return 0;
}
int main(void)
{
    char cs[5][20]={"beautiful","student","China","function","include"};
    int i,ik;
    srand(time(0));
    i=rand()%5;
/***********FOUND***********/
    funtest(i,cs[][50]);
    if(ik==1)
        printf("Right! \n");
    else
        printf("Wrong! The word is:%s\n",cs[i]);
    return 0;
}
