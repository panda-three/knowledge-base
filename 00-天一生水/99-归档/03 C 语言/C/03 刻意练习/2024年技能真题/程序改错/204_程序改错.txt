/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：程序找出一维数组a中的非素数，再转储到二维数组b的左下角。

程序运行结果如下：
18 21 32 25 51 9
18
21 32
25 51 9
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>

int noprime(int x)
{
    int i;
    if(x==2) return 0;
/***********FOUND***********/
    if(x%2) return 1;
    for(i=3;i*i<=x && x%i;i+=2)
        ;
/***********FOUND***********/
    return i*i>=x;
}
int main(void)
{
    int i,j,n,pos;
    int a[]={2,3,18,21,32,25,51,9,31};
    int b[3][3];
    n=sizeof(a)/sizeof(int);
    for(i=j=0;i<n;++i)
        if(noprime(a[i]))
            a[j++]=a[i];
    n=j;
    for(i=0;i<n;++i)
        printf("%3d",a[i]);
    printf("\n");
    for(i=0;i<3;++i)
        for(j=0;j<=i;++j)
        {
/***********FOUND***********/
            pos=(1+j)*i/2;
            b[i][j]=a[pos+j];
        }
    for(i=0;i<3;++i)
    {
        for(j=0;j<=i;++j)
            printf("%3d",b[i][j]);
        printf("\n");
    }
    return 0;
}

答案：

=======(答案1)=======
if(x%2==0)return 1;
=========或=========
if(!(x%2))return 1;

=======(答案2)=======
return i*i<=x;
=========或=========
return x>=i*i;
=========或=========
return i*i-x<=0
=========或=========
return x-i*i>=0

=======(答案3)=======
pos=(1+i)*i/2;
=========或=========
pos=(i+1)*i/2;
=========或=========
pos=i*(1+i)/2;
=========或=========
pos=i*(i+1)/2;
