/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：程序打印N阶方阵，N为奇数，要求行、列、对角线的和均相等，当N为3时，和为15。

程序运行结果如下：
4    3    8
9    5    1
2    7    6

具体方法：
    1、在最后一列的中间填1 ；
    2、在填N的整倍数的下一个数时，填在N的整倍数的左侧；
    3、其他数填在上一个数的右下方，若右下方的列不存在，则填在该行最左列，
       若右下方的行不存在，则填在该列最上行。
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>

#define N 3

void nine(int a[][N])
{
    int i,j,k=1;
/***********FOUND***********/
    i=1;
    j=N-1;
    a[i][j]=k++;
    while(k<=N*N)
    {
/***********FOUND***********/
        if(k%N==0)
            j--;
        else
        {
            i++;j++;
            if(j>N-1) j=0;
            if(i>N-1) i=0;
        }
/***********FOUND***********/
        a[i][j]=k;
    }
}
void print(int a[][N])
{
    int i,j;
    for(i=0;i<N;i++)
    {
        for(j=0;j<N;j++)
            printf("%d\t",a[i][j]);
        printf("\n");
    }
}
int main(void)
{
    int a[N][N]={0};
    nine(a);
    print(a);
    return 0;
}
