/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：下列程序中函数wordsearch返回a数组中回文单词的个数。
      回文单词是指：一个单词自左向右读与自右向左读字母相同(字母不区分大小写)
      例如，Dad是回文单词。

程序运行结果如下：
Dad left home at noon.
发现 2 个回文单词：
Dad
noon
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int wordsearch( char a[] , char b[][10] )
{
    int i,k,m,n,t,c=0,d;
    for(i=0;a[i];i++)
    {
        if(isalpha(a[i]))
        {
            for(k=i;isalpha(a[k]);k++) ;
            m=i;
            n=k-1;
            while(toupper(a[m])==toupper(a[n])&&m<n)
            {
                m++; n--;
            }
        /***********FOUND***********/
            if(m==n)
            {
                for(d=0,t=i;t<k; t++)
                    b[c][d++]=a[t];
                b[c][d]='\0';
                c++;
            }
       /***********FOUND***********/
            i++;
        }
    }
    return c ;
}
int main(void)
{
    char s1[]=" Dad left  home at   noon.  " ;
    char s2[10][10] ;
    int i,j;
    puts(s1);
    j=wordsearch(s1,s2);
    printf("发现 %d 个回文单词：\n",j);
    for(i=0;i<j;i++)
    /***********FOUND***********/
        puts(s2);
    return 0;
}
