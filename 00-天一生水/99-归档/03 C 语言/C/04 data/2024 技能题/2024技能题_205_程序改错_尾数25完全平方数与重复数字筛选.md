/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：下列程序的功能是找出所有满足以下三个条件的3位和4位十进制正整数：
      （1）该数是完全平方数；
      （2）该数的末两位是25；
      （3）组成该数的各位数字中有两位是相同的。

例如：225是满足所给条件的三位正整数（15*15=225）；1225是满足所给条件的
      四位正整数（35*35=1225）。
      函数find中数组x存放的是找到满足条件的正整数，函数返回找到的正整数个数。

程序运行结果如下：
n=3:225
n=4:1225  2025  4225  5625  7225
------------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <math.h>

int find(int n,int x[])
{
    int num=1,k=0,m,i,j,y[10],flag;
    for(i=1;i<n;i++)
/***********FOUND***********/
        num=num*i;
    for(i=num+5;i<num*10;i+=10)
    {
        for(j=0;j<10;j++)
            y[j]=0;
        m=i;
        while(m)
        {
            y[m%10]++;
            m/=10;
        }
        flag=0;
        for(j=0;j<10;j++)
/***********FOUND***********/
            if(y[j]!=0)
                flag=1;
        if(flag==1&&(int)sqrt(i+1)*(int)sqrt(i+1)==i&&i%100==25)
            x[k++]=i;
    }
    return k;
}

int main(void)
{
    int i,j,k,n=3,a[100];
    for(j=0;j<2;j++)
    {
/***********FOUND***********/

        printf("n=%d:",j+n);
        for(i=0;i<k;i++)
            printf("%-6d",a[i]);

    return 0;
}

## 二、标签

- 重要性：重要
- 题型：程序改错题（尾数25完全平方数与重复数字筛选）
- 答案相关考点：
    - 生成 n 位数起始数量级。
    - 统计是否恰有数字出现两次。
    - 按 n 调用查找函数。
- 解题动作：
    - 根据位数构造枚举范围。
    - 筛选末两位为 25 的数。
    - 统计每个数字出现次数。
    - 判断是否存在出现两次的数字。
    - 判断完全平方数。
- 易错点：
    - 数量级用乘 i 而不是乘 10。
    - 把出现过一次误当成重复两位。
    - 调用 find 时没有传 n 位数。
