# 字符串遍历与字符分类练习组

1. 题组定位
    a. 起点题是 [[03 Areas/03 C 语言/C/04 data/2026 年真题/V3/2026真题_75_字符串遍历字符分类与数字累积]]。
    b. 本题组训练的核心是：**逐字符遍历字符串，按字符类别处理并输出结果**。
    c. 题组目标不是讲新知识，而是围绕字符串遍历、字符范围判断和数字字符转换重复练。

2. 核心训练目标
    a. 判断字符串遍历范围。
    b. 判断字符属于数字、字母或其他字符。
    c. 将数字字符转换为数值。
    d. 追踪新字符串、计数器或累加器的变化。
    e. 处理字符串结束符 `\0`。

```c
include<stdio.h>
#include<string.h>
int main()
{
    char c[50]={"1H3ap1py!"};
    char str[50];
    long sum=0;
    int s,i=0;
    printf("string:\n");
    printf("%s\n",c);
    for(s=0;s<strlen(c);s++)
    {
        if((c[s]>='a'&&c[s]<='z')||(c[s]>='A'&&c[s]<='Z'))
        {
            str[i]=c[s];
            i++;
        }
        if(c[s]>='0'&&c[s]<='9')
        {
            sum=10 * sum+c[s]-48;
        }

    printf("word:%s\n",str);
    printf("number:%d\n",sum);
    return 0;
}
```

```c
#include<stdio.h>
int main()
{
    char idf[]="Text_123";
    int count[5]={0};
    int p=0;
    printf("%s\n",idf);
    while(idf[p]!='\0')
    {
        if(idf[p]>='A'&&idf[p]<='Z')count[0]++;
        else if(idf[p]>='a'&&idf[p]<='z')count[1]++;
        else if(idf[p]>='0'&&idf[p]<='9')count[2]++;
        else if(idf[p]=='_')count[3]++;
        else count[4]++;
        p++;
    }
    printf("upper\tlower\tnumber\tunder\n");
    for(p=0;p<4;p++)
        printf("%d\t",count[p]);
    printf("\n");
    if(((idf[0]>='A'&&idf[0]<='Z'||idf[0]>='a'&&idf[0]<='z')&& !count[4])
        printf("Valid\n");
    else
        printf("Invalid\n");
    return(0);
}
```

```c
#include<stdio.h>

{
    char number_d[]="19731204";
    unsigned digit[10]={0},i,c,count=0;
    printf("%s\n",number_d);
    for(i=0;i<strlen(number_d);i++)
    {
        c=number_d[i]-48;
        digit[c]++;
    }
    printf("Missing number:\n");
    for(i=0;i<10;i++)
    {
        if(! digit[i])
        {
            printf("%c",i+48);
            count++;
        }
    }
    printf("\ncount:%d\n",count);
    return 0;
}
```
