- 题目原文：

75. 阅读下列程序，请将运行结果填写在答题卡上相应位置。

```c
#include <stdio.h>
#define N 6

int funs(int n)
{
    int t = 0;
    while (n)
    {
        t += n % 10;
        n = n / 10;
    }
    return t;
}

void fun(int a[])
{
    int i, j, t, si, sj;
    for (i = 1; i < N; i++)
    {
        si = funs(a[i]);
        for (j = 0; j < i; j++)
        {
            sj = funs(a[j]);
            if (si > sj)
            {
                t = a[i];
                a[i] = a[j];
                a[j] = t;
            }
        }
    }
}

int main(void)
{
    int i, a[N] = {43, 35, 67, 12, 18, 50};
    for (i = 0; i < N; i++)
        printf("%4d", a[i]);
    printf("
");

    fun(a);

    for (i = 0; i < N; i++)
    {
        printf("%4d", a[i]);
        if ((i + 1) % 3 == 0)
            printf("
");
    }
    return 0;
}
```
