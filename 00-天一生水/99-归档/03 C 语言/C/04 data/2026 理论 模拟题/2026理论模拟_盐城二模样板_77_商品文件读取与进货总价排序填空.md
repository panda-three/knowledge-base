### 77. 文件读取与按进货总价排序

题干：在 D 盘根目录下，有一个名为 `123.txt` 的文件，文件中有 20 条商品的信息，每条信息包括商品名称、进货单价、进货量。从文件中读取所有商品信息，按每种商品进货总价升序排序后，输出进货价最高的三种商品的名称及进货总价。请完善代码中①~④处内容。

```c
#include <stdio.h>
#include <string.h>
#define N 20

// 按进货总价升序排序
void bubbleSort(char name[][50], float data[][3], int n)
{
    int i, j;
    float tempData[3];
    // 临时存放 单价、数量、总价
    char tempName[50];
    // 临时存放名称

    for (i = 0;
    i < n - 1;
    i++)
    {
        for (j = 0;
        j < n - 1 - i;
        j++)
        {
            if (①)
            {
                strcpy(tempName, name[j]);
                strcpy(name[j], name[j + 1]);
                strcpy(name[j + 1], tempName);

                tempData[0] = data[j][0];
                tempData[1] = data[j][1];
                tempData[2] = data[j][2];

                data[j][0] = data[j + 1][0];
                data[j][1] = data[j + 1][1];
                data[j][2] = data[j + 1][2];

                data[j + 1][0] = tempData[0];
                data[j + 1][1] = tempData[1];
                data[j + 1][2] = tempData[2];
            }
        }
    }
}

int main()
{
    char name[N][50];
    // 商品名称
    float data[N][3];
    // 二维浮点数组：分别存放单价、数量、总价
    FILE *fp;
    int i;

    if ((②) == NULL)
    {
        printf("文件打开失败！\n");
        return 1;
    }

    for (i = 0;
    i < N;
    i++)
    {
        fscanf(fp, "%s %f %f", name[i], &data[i][0], &data[i][1]);
        data[i][2] = data[i][0] * data[i][1];
    }

    fclose(fp);
    bubbleSort(③);

    printf("进货总价最高的三种商品：\n");
    for (④;
    i < N;
    i++)
    {
        printf("名称：%s\t进货总价：%.2f\n", name[i], data[i][2]);
    }

    return 0;
}
```
