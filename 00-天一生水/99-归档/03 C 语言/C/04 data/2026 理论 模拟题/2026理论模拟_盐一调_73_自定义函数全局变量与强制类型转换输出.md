```c
#include <stdio.h>
int cal1(int m, int n)
{return (m - n) + (m * n);
}
int cal2(int x)
{if (x > 10) {
return x/2;}
else
{return x*3;
}
}
int g = 2;
int main()
{int a = 7, b = 4, res1, res2, res3;
double f = 3.5;
res1 = cal1(a, b);
res2 = cal2(res1 + g);
res3 = res1 - res2 + (int)f;
printf("res1=%d,res2=%d\n", res1, res2);
printf("res3=%d,f=%.1f\n", res3, f);
return 0;
}
```
