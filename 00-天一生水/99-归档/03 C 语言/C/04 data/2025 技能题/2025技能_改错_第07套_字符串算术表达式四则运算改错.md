```c
/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：二维字符数组str的每行存放了一个只包含正整数和加(+)、减(-)、乘(*)、
      除(/)等4种运算符的算术表达式。程序根据四则运算规则计算每个表达式的值。
      当有除法运算时，分母不能为0，且运算结果仅保留整数。  
  
说明：表达式仅包含正整数及+、-、*、/四种运算符；除法分母不为0；
      除法仅保留整数部分。

程序运行结果： 
   0:  12-2*3+6=  12
   1:   5-4/2-8=  -5
   2: 3*5/2+4*2=  15   
-----------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define N 100 

int cal(char str[N])
{
    int stk[N]={0}, top ,s,i,sum,n,fh;
    char preSign = '+';
    top=s = 0;
    n = strlen(str);
    for (i = 0; i < n; i++) 
    {
        while (isdigit(str[i])) 
/***********FOUND***********/
            s = s * 10 +str[i] - '0'; 
        if(preSign=='*')
            stk[top-1] *= s;
        else if(preSign=='/')
            stk[top-1] /= s;
        else 
        {
            if(preSign=='+')
                fh=1;
            else
                fh=-1;
/***********FOUND***********/                    
            stk[top-1] = s;
        }    
/***********FOUND***********/             
        preSign = str[i+1];
        s = 0;
    }    
    sum=0;
    for (i = 0; i < top; i++) 
        sum =sum+stk[i];
    return sum;
}

int main(void)
{
    char str[][N]={"12-2*3+6","5-4/2-8","3*5/2+4*2"};
    int i;
    for(i=0;i<3;i++)    
        printf("%4d:%10s=%4d\n",i,str[i],cal(str[i]));
}
```
