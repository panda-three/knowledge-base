```c
/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：n位格雷码是由n个0或1组成的2^n个编码。当n=4时，16个4位格雷码如下：
      0000  0001  0011  0010  0110  0111  0101  0100
      1100  1101  1111  1110  1010  1011  1001  1000

4位格雷码生成规则如下： 
    1.第1个格雷码是0000;
    2.当i>1且是偶数时，则将第i-1个格雷码的末位取反，可得到第i个格雷码；
      当i>1且是奇数时，则将第i-1个格雷码从右往左的方向查找，将遇到的第一
      个1的左边一位取反，可得到第i个格雷码；
    3.不断重复第2步，即可得到全部16个4位格雷码。

程序运行结果如下：
0000    0001    0011    0010
0110    0111    0101    0100
1100    1101    1111    1110
1010    1011    1001    1000
-----------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>
#include <string.h>

#define N 16

void create(char gray[][5])
{
    int i,k;
    for(k=1;;k++)
    {
/***********FOUND***********/
        strcpy(gray[k-1],gray[k]);
        if(k%2)
            gray[k][3]='1'-gray[k][3]+'0';
        else
        {
            i=3;
/***********FOUND***********/
            while(i>=0 || gray[k][i]==1)
                --i;
            if(i>0)
            {
                --i;
                gray[k][i]='1'-gray[k][i]+'0';
            }
            else
                break;
        }
    }
}
int main(void)
{
    int i,j;
    char gray[N][5]={{"0000"}};
    create(gray);
    for(i=0;i<N/4;i++)
    {
        for(j=0;j<N/4;j++)
/***********FOUND***********/
            printf("%s\t",gray[i][j]);
        printf("\n");
    }
    return 0; 
}
```
