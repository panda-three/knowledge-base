```c
/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：一个二维数组中最多存在一个鞍点。如果某元素a[r][c]既是r行的最大元素，
      又是c列的最小元素，则称[r][c]为二维数组a的鞍点。
      程序寻找二维数组a的鞍点。用一维数组max_c[N]保存每行最大元素的列下标，
      min_r[N]保存每列最小元素的行下标。

程序运行结果如下：
存在鞍点：a[3][0]=8
-----------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>

#define N 5

/***********FOUND***********/
int cal(int a[][],int max_c[],int min_r[])
{
    int i,j;
    for(i=0;i<N;i++)
        for(j=0;j<N;j++)
        {
/***********FOUND***********/
            if(a[i][i]>a[i][max_c[j]])
                max_c[i]=j;
            if(a[i][j]<a[min_r[j]][j])
                min_r[j]=i;
        }
} 
int main(void)
{
    int a[N][N]={{11,3,5,6,16},{12,4,7,8,10},{10,5,6,9,11},{8,6,4,7,2},{15,10,11,20,25}};
    int max_c[N]={0},min_r[N]={0};
    int i,bz;
    cal(a,max_c,min_r);
    bz=0;
    for(i=0;i<N;++i)
/***********FOUND***********/
        if(min_r[max_c[i]]=i)
        {
            printf("存在鞍点：a[%d][%d]=%d\n",i,max_c[i],a[i][max_c[i]]);
            bz=1;
        }
    if(bz==0)
        printf("该二维数组不存在鞍点。\n");
    return 0;
}
```
