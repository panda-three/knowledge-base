```c
/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：函数int find(int lft,int rgt,int nums[])的功能是找出[lft,rgt]
      (10=<lft<rgt<=999)中最高位与最低位数字之和为素数的整数存入数组nums中，
      并返回所找到的所有整数的个数。

程序运行结果如下：    
范围[26,28]内没有满足条件的整数！

范围[30,50]内满足条件的数共9个，分别是：
30,32,34,38,41,43,47,49,50

范围[80,113]内满足条件的数共12个，分别是：
83,85,89,92,94,98,101,102,104,106,111,112
-----------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h>

#define N 1000

int find(int lft,int rgt, int nums[])
{
    int n,i,j,tmp,s,p;
    int flg[19];
    for(i=0;i<19;i++)
        flg[i]=1;
    for(i=2;i<19;i++)
    {    
        for(j=2;i*j<19;j++)
/***********FOUND***********/
            flg[j]=1;
    }      
    n=0;
    for(i=lft;i<=rgt;i++)
    {    
        tmp=i;p=1;
/***********FOUND***********/
        while(tmp==0) 
        {    
            p=p*10;    
            tmp=tmp/10;    
        } 
        s=i/p+i%10;
        if(flg[s]!=0&&s>=2)
            nums[n++]=i;
    }   
    return n;
}
    
int main()
{
    int i,j,nums[N],n;
    int fw[3][2]={{26,28},{30,50},{80,113}};
    for(i=0;i<3;i++)
    {
        if(fw[i][0]>fw[i][1] || fw[i][0]<10 ||fw[i][1]>1000 )
            printf("[%d,%d]数据范围不满足题意！\n",fw[i][0],fw[i][1]);
        else
        {  
/***********FOUND***********/
            find(fw[i],fw[i],nums);
            if(n==0)
                printf("范围[%d,%d]内没有满足条件的整数！\n\n",fw[i][0],fw[i][1]); 
            else
            {
                printf("范围[%d,%d]内满足条件的数共%d个，分别是：\n",fw[i][0],fw[i][1],n);
                for(j=0;j<n;j++)
                    printf("%d,",nums[j]);
                printf("\b \n\n");
            }
        }     
    }
    return 0;
}
```
