```c
/*----------------------------------------------------------------------
【程序改错】
------------------------------------------------------------------------
题目：输入一个小于256的自然数，输出： 
      1.该自然数所对应的8位二进制字符串;
      2.该字符串中字符'1'的个数;
      3.判断该字符串中字符'1'的个数是否为素数。

函数int change(int x,char bin[]) 将自然数x所对应的8位二进制字符串保存
在字符数组bin[]中，并返回该字符串中'1'的个数。

例如：
当输入13时，程序运行结果如下：
13对应的8位二进制字符串是00001101，其中有3个1，3是素数。

当输入51时，程序运行结果如下：
51对应的8位二进制字符串是00110011，其中有4个1，4不是素数。

当输入8时，程序运行结果如下：
8对应的8位二进制字符串是00001000，其中有1个1，1不是素数。
-----------------------------------------------------------------------
注意：不得增行或删行，不得更改程序的结构
----------------------------------------------------------------------*/
#include <stdio.h> 

#define N 9

int change(int x,char bin[]) 
{
    int i,len=8,k;
/***********FOUND***********/
    bin[len]=0;
    while(len>=0)
    {
/***********FOUND***********/
        bin[len]=x%2;
        x/=2;
    }
    for(k=i=0;bin[i];i++)
        if(bin[i]=='1')
           k++;
    return k;
}
int main()
{
    char bin[N];
    int i,x,n;
    printf("输入一个正整数（小于等于255）:");
    scanf("%d",&x);
    n=change(x,bin);
    for(i=2;i<n;i++)
        if(n%i==0) 
            break;
/***********FOUND***********/    
    if(i>=n)
        printf("\n%d对应的8位二进制字符串是%s，其中有%d个1，%d是素数。\n",x,bin,n,n);
    else
        printf("\n%d对应的8位二进制字符串是%s，其中有%d个1，%d不是素数。\n",x,bin,n,n);
    return 0;
}
```
